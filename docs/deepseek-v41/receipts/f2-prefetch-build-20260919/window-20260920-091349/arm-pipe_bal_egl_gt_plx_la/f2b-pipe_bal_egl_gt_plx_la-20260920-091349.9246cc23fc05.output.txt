@dataclass(frozen=True)
class Prompt:
    id: str
    text: str
    max_tokens: int = 256
    tags: tuple[str, ...] = ()
    metadata: Mapping[str, Any] = field(default_factory=dict)

    def to_json(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "text": self.text,
            "max_tokens": self.max_tokens,
            "tags": list(self.tags),
            "metadata": dict(self.metadata),
        }

    @classmethod
    def from_json(cls, row: Mapping[str, Any]) -> "Prompt":
        tags = row.get("tags", ())
        if isinstance(tags, str):
            tags = (tags,)
        return cls(
            id=str(row["id"]),
            text=str(row["text"]),
            max_tokens=int(row.get("max_tokens", 256)),
            tags=tuple(str(t) for t in tags),
            metadata=dict(row.get("metadata", {})),
        )


@dataclass(frozen=True)
class PromptSet:
    prompts: tuple[Prompt, ...]
    source: str = "<memory>"

    def __len__(self) -> int:
        return len(self.prompts)

    def __iter__(self) -> Iterator[Prompt]:
        return iter(self.prompts)

    def digest(self) -> str:
        h = hashlib.sha256()
        for p in self.prompts:
            h.update(p.id.encode("utf-8"))
            h.update(b"\x00")
            h.update(p.text.encode("utf-8"))
            h.update(b"\x00")
            h.update(str(p.max_tokens).encode("ascii"))
            h.update(b"\x00")
            for t in p.tags:
                h.update(t.encode("utf-8"))
                h.update(b"\x01")
            h.update(b"\x02")
        return h.hexdigest()


def load_prompts_jsonl(path: Path | str) -> PromptSet:
    p = Path(path)
    prompts: list[Prompt] = []
    with p.open("r", encoding="utf-8") as fh:
        for lineno, line in enumerate(fh, start=1):
            line = line.strip()
            if not line:
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(f"{p}:{lineno}: invalid JSON: {exc}") from exc
            if not isinstance(row, Mapping):
                raise ValueError(f"{p}:{lineno}: row must be a JSON object")
            prompts.append(Prompt.from_json(row))
    return PromptSet(prompts=tuple(prompts), source=str(p))


def write_prompts_jsonl(path: Path | str, prompts: Iterable[Prompt]) -> int:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    count = 0
    with p.open("w", encoding="utf-8") as fh:
        for prompt in prompts:
            fh.write(json.dumps(prompt.to_json(), sort_keys=True))
            fh.write("\n")
            count += 1
    return count
