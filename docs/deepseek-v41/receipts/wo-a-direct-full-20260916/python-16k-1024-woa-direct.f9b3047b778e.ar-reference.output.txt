+def _percentile(values: Sequence[float], q: float) -> float | None:
+    """Linear-interpolation percentile over a non-empty sequence.
+
+    Returns None for an empty input so callers can omit the field rather
+    than fabricate a zero. ``q`` is a fraction in [0, 1].
+    """
+
+    if not values:
+        return None
+    ordered = sorted(float(value) for value in values)
+    if len(ordered) == 1:
+        return ordered[0]
+    position = max(0.0, min(1.0, float(q))) * (len(ordered) - 1)
+    lower = int(position)
+    upper = min(lower + 1, len(ordered) - 1)
+    weight = position - lower
+    return ordered[lower] * (1.0 - weight) + ordered[upper] * weight
